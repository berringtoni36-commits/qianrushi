## 一、本质理解

图的本质是 **节点的集合 + 节点之间的连接关系**。

labuladong 将图归为"链表的延伸"——树的每个节点有多个子节点（多叉树），而图就是多叉树的进一步泛化（可以有环）。

**核心关系：**
```
链表 → 二叉树 → 多叉树 → 图
（线性） （二叉）（多叉） （有环）
```

## 三、核心模板

### 模板 1：图 DFS

```cpp
void dfsGraph(const vector<vector<int>>& graph, int start, vector<bool>& visited) {
    visited[start] = true;
    for (int next : graph[start]) {
        if (!visited[next]) dfsGraph(graph, next, visited);
    }
}
```
### 模板 2：图 BFS（求最短路径/最少步数）

```cpp
vector<int> bfsDistance(const vector<vector<int>>& graph, int start) {
    vector<int> distance(graph.size(), -1);
    queue<int> q;
    q.push(start);
    distance[start] = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : graph[u]) if (distance[v] == -1) {
            distance[v] = distance[u] + 1;
            q.push(v);
        }
    }
    return distance;
}
```
### 模板 3：网格 DFS（岛屿问题）

```cpp
class Solution {
public:
    int numIslands(vector<vector<char>>& grid) {
        if (grid.empty() || grid[0].empty()) return 0;
        int rows = grid.size(), columns = grid[0].size(), answer = 0;
        auto sink = [&](auto&& self, int x, int y) -> void {
            if (x < 0 || x >= rows || y < 0 || y >= columns || grid[x][y] != '1') return;
            grid[x][y] = '0';
            self(self, x + 1, y); self(self, x - 1, y);
            self(self, x, y + 1); self(self, x, y - 1);
        };
        for (int i = 0; i < rows; ++i)
            for (int j = 0; j < columns; ++j)
                if (grid[i][j] == '1') ++answer, sink(sink, i, j);
        return answer;
    }
};
```
### 模板 4：多源 BFS（腐烂的橘子）

```cpp
class Solution {
public:
    int orangesRotting(vector<vector<int>>& grid) {
        int fresh = 0, minutes = 0;
        queue<pair<int, int>> q;
        for (int i = 0; i < grid.size(); i ++)
            for (int j = 0; j < grid[0].size(); j ++)
                if (grid[i][j] == 2) q.push({i, j});
                else if (grid[i][j] == 1) fresh ++;
        const int d[5] = {-1, 0, 1, 0, -1};
        while (fresh && !q.empty()) {
            minutes ++;
            for (int size = q.size(); size --; ) {
                auto [i, j] = q.front(); q.pop();
                for (int k = 0; k < 4; k ++) {
                    int x = i + d[k], y = j + d[k + 1];
                    if (x < 0 || x == grid.size() || y < 0 || y == grid[0].size() || grid[x][y] != 1) continue;
                    grid[x][y] = 2;
                    fresh --;
                    q.push({x, y});
                }
            }
        }
        return fresh ? -1 : minutes;
    }
};
```
### 模板 5：拓扑排序（Kahn 算法 / BFS 版）

```cpp
bool canFinishTemplate(int courseCount, const vector<vector<int>>& prerequisites) {
    vector<vector<int>> graph(courseCount);
    vector<int> indegree(courseCount);
    for (const auto& edge : prerequisites) {
        graph[edge[1]].push_back(edge[0]);
        ++indegree[edge[0]];
    }
    queue<int> q;
    for (int i = 0; i < courseCount; ++i) if (indegree[i] == 0) q.push(i);
    int learned = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        ++learned;
        for (int v : graph[u]) if (--indegree[v] == 0) q.push(v);
    }
    return learned == courseCount;
}
```

## 四、Trie（前缀树）

```cpp
class Trie {
    struct Node {
        array<unique_ptr<Node>, 26> child{};
        bool terminal = false;
    };
    Node root;

public:
    void insert(const string& word) {
        Node* node = &root;
        for (char c : word) {
            auto& next = node->child[c - 'a'];
            if (!next) next = make_unique<Node>();
            node = next.get();
        }
        node->terminal = true;
    }

    bool search(const string& word) const {
        const Node* node = &root;
        for (char c : word) {
            const auto& next = node->child[c - 'a'];
            if (!next) return false;
            node = next.get();
        }
        return node->terminal;
    }
};
```

## 六、易错点与技巧

1. **DFS vs BFS 选择**：求最短路径/最少步数用 BFS，求连通性/路径用 DFS
2. **visited 的两种方式**：额外 visited 集合（通用）vs 原地修改（网格题"沉岛"）
3. **多源 BFS**：腐烂橘子是典型，多个起点同时入队，逐层扩散
4. **拓扑排序的两种实现**：BFS（Kahn）和 DFS（检测后序遍历的逆序），Kahn 更直观
5. **Trie 的内存**：每个字符一个节点，字符串量大时内存消耗大

## 七、复杂度总结

| 操作 | 时间复杂度 | 空间复杂度 |
|------|-----------|-----------|
| 图 DFS/BFS | O(V + E) | O(V) |
| 网格 DFS | O(m × n) | O(m × n) |
| 拓扑排序 | O(V + E) | O(V + E) |
| Trie 操作 | O(len(word)) | O(总字符数) |

### 网格 DFS

```cpp
void floodFill(vector<vector<char>>& grid, int x, int y) {
    int rows = grid.size(), cols = grid[0].size();
    if (x < 0 || x >= rows || y < 0 || y >= cols || grid[x][y] != '1') return;
    grid[x][y] = '0';
    static constexpr int dx[4] = {1, -1, 0, 0};
    static constexpr int dy[4] = {0, 0, 1, -1};
    for (int d = 0; d < 4; ++d) floodFill(grid, x + dx[d], y + dy[d]);
}
```

### 多源 BFS

```cpp
int multiSourceBfs(vector<vector<int>>& grid, queue<pair<int, int>> q) {
    int rows = grid.size(), cols = grid[0].size(), minutes = 0;
    static constexpr int dx[4] = {1, -1, 0, 0};
    static constexpr int dy[4] = {0, 0, 1, -1};
    while (!q.empty()) {
        int levelSize = q.size();
        while (levelSize--) {
            auto [x, y] = q.front(); q.pop();
            for (int d = 0; d < 4; ++d) {
                int nx = x + dx[d], ny = y + dy[d];
                if (nx < 0 || nx >= rows || ny < 0 || ny >= cols || grid[nx][ny] != 1) continue;
                grid[nx][ny] = 2;
                q.push({nx, ny});
            }
        }
        if (!q.empty()) ++minutes;
    }
    return minutes;
}
```

### Kahn 拓扑排序

```cpp
bool canFinishTemplate(int n, const vector<vector<int>>& edges) {
    vector<vector<int>> graph(n);
    vector<int> indegree(n);
    for (const auto& edge : edges) {
        graph[edge[1]].push_back(edge[0]);
        ++indegree[edge[0]];
    }
    queue<int> q;
    for (int i = 0; i < n; ++i) if (indegree[i] == 0) q.push(i);
    int visited = 0;
    while (!q.empty()) {
        int node = q.front(); q.pop();
        ++visited;
        for (int next : graph[node]) if (--indegree[next] == 0) q.push(next);
    }
    return visited == n;
}
```

- **适用题：** 200、207、208、994。
- **复杂度：** DFS/BFS 为 `O(V+E)`；Trie 单次操作为 `O(L)`。
- **易错点：** 入队时立即标记，避免重复入队；拓扑排序的边方向必须与题意一致。
