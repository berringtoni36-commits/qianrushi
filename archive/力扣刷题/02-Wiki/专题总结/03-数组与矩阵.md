## 一、本质理解

**数组的本质是连续内存空间**，支持 O(1) 随机访问，但插入/删除需要 O(n) 移动元素。  
**矩阵是二维数组**，底层也是连续存储（行优先）。

## 二、数组核心技巧

### 1. 前缀和（区间和问题）

```cpp
vector<long long> buildPrefixSum(const vector<int>& nums) {
    vector<long long> prefix(nums.size() + 1);
    for (int i = 0; i < static_cast<int>(nums.size()); ++i) {
        prefix[i + 1] = prefix[i] + nums[i];
    }
    return prefix;
}

long long rangeSum(const vector<long long>& prefix, int left, int right) {
    return prefix[right + 1] - prefix[left];
}
```
**适用场景：** 需要频繁计算子数组和 → [[01-哈希表|01-哈希表]] 模板 4

### 2. 差分（区间增减问题）

```cpp
vector<long long> applyRangeAdds(int n, const vector<array<int, 3>>& updates) {
    vector<long long> difference(n + 1);
    for (auto [left, right, delta] : updates) {
        difference[left] += delta;
        if (right + 1 < n) difference[right + 1] -= delta;
    }
    for (int i = 1; i < n; ++i) difference[i] += difference[i - 1];
    difference.resize(n);
    return difference;
}
```
### 3. 原地操作（空间 O(1)）

**三次翻转法 — 轮转数组：**

```cpp
void rotateArray(vector<int>& nums, int k) {
    if (nums.empty()) return;
    k %= nums.size();
    reverse(nums.begin(), nums.end());
    reverse(nums.begin(), nums.begin() + k);
    reverse(nums.begin() + k, nums.end());
}
```
**前缀积 + 后缀积 — 除自身以外数组的乘积：**

```cpp
vector<int> productExceptSelfTemplate(const vector<int>& nums) {
    vector<int> answer(nums.size(), 1);
    int prefix = 1;
    for (int i = 0; i < static_cast<int>(nums.size()); ++i) {
        answer[i] = prefix;
        prefix *= nums[i];
    }
    int suffix = 1;
    for (int i = static_cast<int>(nums.size()) - 1; i >= 0; --i) {
        answer[i] *= suffix;
        suffix *= nums[i];
    }
    return answer;
}
```
**原地哈希 — 缺失的第一个正数：**

```cpp
int firstMissingPositiveTemplate(vector<int> nums) {
    int n = nums.size();
    for (int i = 0; i < n; ++i) {
        while (nums[i] >= 1 && nums[i] <= n && nums[nums[i] - 1] != nums[i]) {
            swap(nums[i], nums[nums[i] - 1]);
        }
    }
    for (int i = 0; i < n; ++i) if (nums[i] != i + 1) return i + 1;
    return n + 1;
}
```
### 4. Kadane 算法（最大子数组和）

```cpp
class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int ans = nums[0], current = 0;
        for (int x : nums) {
            current = max(x, current + x);
            ans = max(ans, current);
        }
        return ans;
    }
};
```

## 三、矩阵核心技巧

### 1. 螺旋遍历（边界收缩法）

```cpp
vector<int> spiralOrderTemplate(const vector<vector<int>>& matrix) {
    if (matrix.empty() || matrix[0].empty()) return {};
    vector<int> answer;
    int top = 0, bottom = matrix.size() - 1;
    int left = 0, right = matrix[0].size() - 1;
    while (top <= bottom && left <= right) {
        for (int j = left; j <= right; ++j) answer.push_back(matrix[top][j]);
        ++top;
        for (int i = top; i <= bottom; ++i) answer.push_back(matrix[i][right]);
        --right;
        if (top <= bottom)
            for (int j = right; j >= left; --j) answer.push_back(matrix[bottom][j]);
        --bottom;
        if (left <= right)
            for (int i = bottom; i >= top; --i) answer.push_back(matrix[i][left]);
        ++left;
    }
    return answer;
}
```
### 2. 旋转图像（转置 + 翻转）

```cpp
class Solution {
public:
    void rotate(vector<vector<int>>& matrix) {
        int n = matrix.size();
        for (int i = 0; i < n; i ++)
            for (int j = i + 1; j < n; j ++)
                swap(matrix[i][j], matrix[j][i]);
        for (auto& row : matrix) reverse(row.begin(), row.end());
    }
};
```
### 3. 矩阵置零（原地标记法）

```cpp
class Solution {
public:
    void setZeroes(vector<vector<int>>& matrix) {
        int m = matrix.size(), n = matrix[0].size();
        bool firstCol = false;
        for (int i = 0; i < m; i ++) {
            if (matrix[i][0] == 0) firstCol = true;
            for (int j = 1; j < n; j ++)
                if (matrix[i][j] == 0) matrix[i][0] = matrix[0][j] = 0;
        }
        for (int i = m - 1; i >= 0; i --) {
            for (int j = n - 1; j >= 1; j --)
                if (matrix[i][0] == 0 || matrix[0][j] == 0) matrix[i][j] = 0;
            if (firstCol) matrix[i][0] = 0;
        }
    }
};
```
### 4. 搜索二维矩阵（右上角搜索法）

```cpp
class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        int m = matrix.size(), n = matrix[0].size();
        int left = 0, right = m * n - 1;
        while (left <= right) {
            int mid = left + (right - left) / 2;
            int x = matrix[mid / n][mid % n];
            if (x == target) return true;
            if (x < target) left = mid + 1;
            else right = mid - 1;
        }
        return false;
    }
};
```

## 五、易错点与技巧

1. **边界条件**：矩阵题中始终注意 `row` 和 `col` 的范围，以及空矩阵的特殊情况
2. **原地操作**：想修改原数组时，先问自己"会不会覆盖还没用的数据"
3. **取模运算**：轮转数组 `k %= len(nums)` 处理 k > n 的情况
4. **原地哈希**：用数组本身的索引作为哈希 key，值作为标记，空间 O(1)

## YXC 模板精编

### 快速排序原型

```cpp
void quickSort(vector<int>& a, int left, int right) {
    if (left >= right) return;
    int i = left - 1, j = right + 1;
    int pivot = a[left + (right - left) / 2];
    while (i < j) {
        do ++i; while (a[i] < pivot);
        do --j; while (a[j] > pivot);
        if (i < j) swap(a[i], a[j]);
    }
    quickSort(a, left, j);
    quickSort(a, j + 1, right);
}
```

### 快速选择（第 K 大）

```cpp
int quickSelect(vector<int>& a, int left, int right, int target) {
    if (left == right) return a[left];
    int i = left - 1, j = right + 1;
    int pivot = a[left + (right - left) / 2];
    while (i < j) {
        do ++i; while (a[i] < pivot);
        do --j; while (a[j] > pivot);
        if (i < j) swap(a[i], a[j]);
    }
    if (target <= j) return quickSelect(a, left, j, target);
    return quickSelect(a, j + 1, right, target);
}

int findKthLargest(vector<int>& nums, int k) {
    return quickSelect(nums, 0, static_cast<int>(nums.size()) - 1,
                       static_cast<int>(nums.size()) - k);
}
```

### 区间合并

```cpp
vector<vector<int>> mergeIntervals(vector<vector<int>> intervals) {
    sort(intervals.begin(), intervals.end());
    vector<vector<int>> answer;
    for (const auto& interval : intervals) {
        if (answer.empty() || answer.back()[1] < interval[0])
            answer.push_back(interval);
        else
            answer.back()[1] = max(answer.back()[1], interval[1]);
    }
    return answer;
}
```

- **适用题：** [[56-合并区间|56]]、[[75-颜色分类|75]]、[[215-数组中的第K个最大元素|215]]。
- **复杂度：** 快排平均 `O(n log n)`；快速选择平均 `O(n)`；区间合并 `O(n log n)`。
- **易错点：** Hoare 分区递归边界必须是 `[left, j]` 与 `[j+1, right]`；第 K 大转换为升序下标 `n-k`。
